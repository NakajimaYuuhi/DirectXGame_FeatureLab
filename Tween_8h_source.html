87// Property Tween Implementation
88template <typename T>
89class CTweenProperty : public CTween
90{
91public:
92 using GetterFunc = std::function<T()>;
93 using SetterFunc = std::function<void(const T&)>;
94
95 CTweenProperty(GetterFunc getter, SetterFunc setter, T endValue, float duration)
96 : m_getter(getter), m_setter(setter), m_endValue(endValue)
97 {
98 m_duration = (duration > 0.0f) ? duration : 0.0001f;
99 if (m_getter)
100 {
101 m_startValue = m_getter();
102 }
103 }
104
105 virtual void Update(float deltaTime) override
106 {
107 if (m_isPaused || m_isKilled || m_isComplete) return;
108
109 // 1. Handle Delay
110 if (m_delayTimer < m_delay)
111 {
112 m_delayTimer += deltaTime;
113 if (m_delayTimer < m_delay) return;
114 // Delay finished, capture start value fresh
115 if (m_getter) m_startValue = m_getter();
116 }
117
118 // 2. Advance time
119 m_elapsedTime += deltaTime;
120 float progress = m_elapsedTime / m_duration;
121 bool loopFinished = false;
122
123 if (progress >= 1.0f)
124 {
125 progress = 1.0f;
126 loopFinished = true;
127 }
128
129 // 3. Evaluate ease
130 float easeT = EaseUtility::Evaluate(m_ease, progress);
131 if (m_isBackwards)
132 {
133 easeT = 1.0f - easeT;
134 }
135
136 // 4. Apply interpolated value
137 T currentVal = TweenMath::Lerp(m_startValue, m_endValue, easeT);
138 if (m_setter)
139 {
140 m_setter(currentVal);
141 }
142
143 if (m_onUpdate)
144 {
145 m_onUpdate();
146 }
147
148 // 5. Handle loop / completion
149 if (loopFinished)
150 {
151 m_completedLoops++;
152 if (m_loops != -1 && m_completedLoops >= m_loops)
153 {
154 m_isComplete = true;
155 if (m_onComplete)
156 {
157 m_onComplete();
158 }
159 }
160 else
161 {
162 // Loop restart or yoyo
163 m_elapsedTime = 0.0f;
164 if (m_loopType == LoopType::Yoyo)
165 {
166 m_isBackwards = !m_isBackwards;
167 }
168 }
169 }
170 }
171
172 virtual bool IsComplete() const override { return m_isComplete; }
173 virtual bool IsKilled() const override { return m_isKilled; }
174 virtual void Kill() override { m_isKilled = true; }
175
176private:
177 GetterFunc m_getter;
178 SetterFunc m_setter;
179 T m_startValue{};
180 T m_endValue{};
181};
